-- ============================================================
-- Schemat bazy danych: sklep e-commerce
-- Kolejność ma znaczenie! Tabele bez zależności są tworzone pierwsze.
-- ============================================================

-- Wymagane do gen_random_uuid()
CREATE EXTENSION IF NOT EXISTS pgcrypto;

-- ------------------------------------------------------------
-- 1. users
-- ------------------------------------------------------------
CREATE TABLE users(
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  email VARCHAR(255) UNIQUE NOT NULL,
  password_hash VARCHAR(255) NOT NULL,
  lastName VARCHAR(255) NOT NULL,
  firstName VARCHAR(255) NOT NULL,
  phone VARCHAR(255) NOT NULL
);

-- ------------------------------------------------------------
-- 2. addresses (zależy od users)
-- ------------------------------------------------------------
CREATE TABLE addresses(
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  street VARCHAR(255) NOT NULL,
  street_number VARCHAR(255) NOT NULL,
  city VARCHAR(255) NOT NULL,
  zip_code VARCHAR(255) NOT NULL,
  country VARCHAR(255) NOT NULL,
  label VARCHAR(255) NOT NULL,
  is_default BOOLEAN NOT NULL
);

CREATE INDEX idx_addresses_user_id ON addresses(user_id);

-- ------------------------------------------------------------
-- 3. categories (self-reference, brak zależności zewnętrznych)
-- ------------------------------------------------------------
CREATE TABLE categories(
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  name VARCHAR(255) NOT NULL,
  photo VARCHAR(255),
  parent_category_id UUID REFERENCES categories(id) ON DELETE SET NULL
);

CREATE INDEX idx_categories_parent_id ON categories(parent_category_id);

-- ------------------------------------------------------------
-- 4. products (zależy od categories)
-- ------------------------------------------------------------
CREATE TABLE products(
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  name VARCHAR(255) NOT NULL,
  photo VARCHAR(255),
  category_id UUID REFERENCES categories(id) ON DELETE SET NULL,
  description VARCHAR(255),
  price_cents INTEGER NOT NULL,
  stock_quantity INTEGER NOT NULL CHECK (stock_quantity >= 0)
);

CREATE INDEX idx_products_category_id ON products(category_id);

-- ------------------------------------------------------------
-- 5. order_status (typ ENUM, musi istnieć przed orders)
-- ------------------------------------------------------------
CREATE TYPE order_status AS ENUM ('WAITING_FOR_PAY', 'PAID', 'IN_PROGRESS', 'DELIVERED', 'CANCELED');

-- ------------------------------------------------------------
-- 6. orders (zależy od users, addresses, order_status)
-- ------------------------------------------------------------
CREATE TABLE orders(
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  address_id UUID NOT NULL REFERENCES addresses(id) ON DELETE RESTRICT,
  user_id UUID NOT NULL REFERENCES users(id) ON DELETE RESTRICT,
  status order_status NOT NULL DEFAULT 'WAITING_FOR_PAY'
);

CREATE INDEX idx_orders_user_id ON orders(user_id);
CREATE INDEX idx_orders_address_id ON orders(address_id);

-- ------------------------------------------------------------
-- 7. order_items (zależy od orders, products)
-- ------------------------------------------------------------
CREATE TABLE order_items(
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  order_id UUID NOT NULL REFERENCES orders(id) ON DELETE CASCADE,
  product_id UUID NOT NULL REFERENCES products(id) ON DELETE RESTRICT,
  quantity INTEGER NOT NULL CHECK (quantity > 0),
  price_product INTEGER NOT NULL
);

CREATE INDEX idx_order_items_order_id ON order_items(order_id);
CREATE INDEX idx_order_items_product_id ON order_items(product_id);
